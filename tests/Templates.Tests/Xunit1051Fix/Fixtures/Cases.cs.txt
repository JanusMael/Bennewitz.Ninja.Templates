using Xunit;

public sealed class TokenCases
{
    [Fact]
    public async Task Calls()
    {
        string path = Path.GetTempFileName();
        await File.WriteAllTextAsync(path, "x");             // the token is the next parameter
        string text = await File.ReadAllTextAsync(path);     // the same, one argument
        await Service.RunAsync(path);                        // an optional parameter comes first: named
        await Service.RunAsync(path, progress: null);        // already named: the token is named too
        await Task.WhenAny(Task.Delay(TimeSpan.FromMilliseconds(1)));  // a call passed as an argument
        File.Delete(path);
        Assert.Equal("x", text);
    }
}

public static class Service
{
    public static Task RunAsync(string path, IProgress<int>? progress = null, CancellationToken ct = default) =>
        Task.CompletedTask;
}

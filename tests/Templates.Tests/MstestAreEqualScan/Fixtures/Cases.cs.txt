using System.Collections.Immutable;
using System.Text.Json.Nodes;
using Microsoft.VisualStudio.TestTools.UnitTesting;

public static class Cases
{
    public static void All()
    {
        var list = new List<int> { 1 };
        int[] array = [1];
        ImmutableArray<int> immutable = [1];
        var json = new JsonArray(1);
        object boxed = list;
        object? other = null;
        object number = 1.0;

        Assert.AreEqual(list, list);                  // COLLECTION-REF: MSTest compares the reference
        Assert.AreEqual(array, array);                // COLLECTION-REF
        Assert.AreEqual(immutable, immutable);        // COLLECTION-EQ: ImmutableArray overrides Equals
        Assert.AreEqual(json, json);                  // COLLECTION-REF
        Assert.AreEqual<object>(list, list);          // COLLECTION-REF: an opaque bound type, a collection argument
        Assert.AreEqual(boxed, other);                // OPAQUE: nothing static says what these are
        Assert.AreEqual<object>("a", boxed);          // SCALAR: only an equal string can pass
        Assert.AreEqual(1.0, (double)number);         // SCALAR: the bound type is double
        Assert.AreEqual("a", "a");                    // SCALAR
        Assert.AreNotEqual(list, new List<int>());    // COLLECTION-REF, but AreNotEqual only gets stricter
        CollectionAssert.AreEqual(list, list);        // not reported: element-wise on both frameworks
    }
}
